function tmdbKey(){
  var direct=(localStorage.getItem('mongul_tmdb_key')||'').trim();
  if(direct)return direct;
  /* 예전 버전에서 저장했을 수 있는 인증값도 그대로 복구 */
  var aliases=['mongul_tmdb_token','tmdb_access_token','tmdb_token','tmdb_api_key','TMDB_API_KEY','tmdbKey','tmdbToken'];
  for(var i=0;i<aliases.length;i++){var v=(localStorage.getItem(aliases[i])||'').trim();if(v){localStorage.setItem('mongul_tmdb_key',v);return v}}
  try{
    for(var j=0;j<localStorage.length;j++){
      var k=localStorage.key(j)||'';var v2=(localStorage.getItem(k)||'').trim();
      if(!v2)continue;
      if(/tmdb|movie.*api|api.*movie/i.test(k) && v2.length>=20){localStorage.setItem('mongul_tmdb_key',v2);return v2}
    }
  }catch(e){}
  return '';
}

async function tmdbFetch(path){
  const key=tmdbKey();
  if(!key)throw new Error('NO_TMDB_KEY');
  const base='https://api.themoviedb.org/3'+path;
  /* API Read Access Token은 Bearer, 구형 API Key는 query parameter로 사용 */
  const looksLikeBearer=key.length>40 || /^(eyJ|v4\.|[A-Za-z0-9_-]{50,})/.test(key);
  let url=base, opts={headers:{accept:'application/json'}};
  if(looksLikeBearer){opts.headers.Authorization='Bearer '+key}
  else{url=base+(path.includes('?')?'&':'?')+'api_key='+encodeURIComponent(key)}
  let r=await fetch(url,opts);
  /* 저장된 인증 형식이 애매한 경우 두 방식을 한 번 더 시도 */
  if(!r.ok && looksLikeBearer){
    const retry=base+(path.includes('?')?'&':'?')+'api_key='+encodeURIComponent(key);
    r=await fetch(retry,{headers:{accept:'application/json'}});
  }else if(!r.ok && !looksLikeBearer){
    r=await fetch(base,{headers:{accept:'application/json',Authorization:'Bearer '+key}});
  }
  if(!r.ok)throw new Error('TMDB_'+r.status);
  return r.json();
}

  if(localStorage.getItem('mongul_effect')!=='1')return;
  try{
    if(!mongulEffectCtx) mongulEffectCtx=new(window.AudioContext||window.webkitAudioContext)();
    var ctx=mongulEffectCtx;
    var start=function(){
      var t=ctx.currentTime;
      var buffer=ctx.createBuffer(1,Math.floor(ctx.sampleRate*.055),ctx.sampleRate);
      var data=buffer.getChannelData(0);
      for(var i=0;i<data.length;i++){
        var env=1-i/data.length;
        data[i]=(Math.random()*2-1)*env*env*.22;
      }
      var src=ctx.createBufferSource(), hp=ctx.createBiquadFilter(), gain=ctx.createGain();
      hp.type='highpass';hp.frequency.value=1400;hp.Q.value=.45;
      gain.gain.setValueAtTime(.0001,t);gain.gain.exponentialRampToValueAtTime(.12,t+.006);
      gain.gain.exponentialRampToValueAtTime(.0001,t+.045);
      src.buffer=buffer;src.connect(hp).connect(gain).connect(ctx.destination);src.start(t);
      /* 아주 짧은 두 번의 마찰감 = 가벼운 연필 사각 소리 */
      setTimeout(function(){
        try{
          var t2=ctx.currentTime, s2=ctx.createBufferSource(), g2=ctx.createGain();

    else start();
  }catch(e){}
};
